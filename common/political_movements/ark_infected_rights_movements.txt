movement_anti_infectors = {
	category = movement_category_ideological

	ideology = ideology_infectors_rejection_movement
	character_ideologies = {
		ideology_infectors_rejection_leader
		ideology_infectors_ignorism_leader
	}

	disband_trigger = {
		has_law = law_type:law_no_infected_rights
	}
	on_disbanded = {}

	creation_trigger = {
		NOT = { has_law = law_type:law_no_infected_rights }
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				has_law = law_type:law_full_infected_rights
			}
			multiply = {
				value = 10
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_infectors_rejection
			has_ideology = ideology:ideology_infectors_ignorism
			is_interest_group_type = ig_landowners
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				has_ideology = ideology:ideology_infectors_rejection
			}
			multiply = {
				value = 5
			}
		}
	}

	pop_support_factors = {
		movement_support_high_urbanization
		movement_support_low_literacy
		movement_support_aristocrats
		movement_support_clergymen
		movement_support_bureaucrats
		movement_support_clerks
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		add = {
			desc = "URBAN_STATE"
			value = 3
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = aristocrats
				}
				add = {
					value = 12
					desc = "POP_ARISTOCRATS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 25
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = clergymen
				}
				add = {
					value = 25
					desc = "POP_CLERGYMEN"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 15
					desc = "POP_BUREAUCRATS"
				}
			}
			else = {
				add = {
					value = 10
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 10
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = slaves
				}
				add = {
					value = -100
					desc = "POP_SLAVES"
				}
			}
			else = {
				add = {
					value = 3
					desc = "LOWER_NO_ICON"
				}
			}
		}

		if = {
			limit = {
				has_pop_culture = infected
			}
			multiply = {
				value = 0
				desc = "INFECTED_POP"
			}
		}


		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1

			subtract = {
				value = state_urbanization_rate
				multiply = 0.5
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = state_urbanization_rate
		}
	}

	additional_radicalism_factors = {
		owner = {
			if = {
				limit = { has_institution = institution_social_security }

				add = {
					value = -3
					subtract = "institution_investment_level(institution_social_security)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_social_security"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "NO_INSTITUTION_social_security"
				}
			}

			if = {
				limit = { has_institution = institution_workplace_safety }

				add = {
					value = -3
					subtract = "institution_investment_level(institution_workplace_safety)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_workplace_safety"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "NO_INSTITUTION_workplace_safety"
				}
			}
		}
	}
}
