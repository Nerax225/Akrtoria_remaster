is_break_trigger = {
	OR = {
		AND = {
			exists = scope:character
			scope:character = {
				OR = {
					has_trait = senile
					has_trait = wounded
					has_trait = syphilis
					has_trait = opium_addiction
					has_trait = alcoholic
					has_trait = shellshocked
					has_trait = cocaine_addiction
					has_trait = kidney_stones
					has_trait = scarred
					has_trait = sickly
				}
			}
		}
		AND = {
			exists = scope:pop
			scope:pop = {
				OR = {
					is_pop_type = slaves
					standard_of_living < 16
				}
			}
		}
	}
}

## Tradition Triggers
aslan_tradition_trigger = {
	OR = {
		AND = {
			exists = scope:character
			scope:character = {
				culture = {
					has_discrimination_trait = tradition_aslan
				}
			}
		}
		AND = {
			exists = scope:pop
			scope:pop = {
				culture = {
					has_discrimination_trait = tradition_aslan
				}
			}
		}
	}
