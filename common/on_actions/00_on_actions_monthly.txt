infectedgen_state = {
	effect = {
		every_scope_state = {
			save_scope_value_as = {
				name = infected_conversion_rate
				value = modifier:state_infected_conversion_add
			}
			save_scope_value_as = {
				name = catastrophe_infected_conversion_add_rate
				value = modifier:catastrophe_infected_conversion_add
			}
			save_scope_value_as = {
				name = state_originium_dust_conversion
				value = modifier:dust_infected_conversion_add
			}
			every_scope_pop = {
				if = {
					limit = {
						AND = {
							NOT = {
								has_pop_culture = infected
							}
							pop_employment_building = building_originmum_mine					
						}

					}
					change_pop_culture = {
						target = cu:infected
						value = scope:infected_conversion_rate
					}
				}
				if = {
					limit = {
						AND = {
							NOT = {
								has_pop_culture = infected
							}					
						}

					}
					change_pop_culture = {
						target = cu:infected
						value = scope:state_originium_dust_conversion
					}
				}
				if = {
					limit = {
						AND = {
							NOT = {
								has_pop_culture = infected
							}
							catastrophe_infected_conversion_add_rate > 0	
						}

					}
					change_pop_culture = {
						target = cu:infected
						value = scope:catastrophe_infected_conversion_add_rate
					}
				}									
			}
		}
	}
}
